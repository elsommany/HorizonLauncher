package com.horizon.launcher

import android.app.Activity
import android.content.Intent
import android.graphics.Bitmap
import android.graphics.Canvas
import android.graphics.Color
import android.graphics.drawable.BitmapDrawable
import android.graphics.drawable.Drawable
import android.net.Uri
import android.os.Bundle
import android.provider.Settings
import android.view.KeyEvent
import android.webkit.JavascriptInterface
import android.webkit.WebResourceRequest
import android.webkit.WebResourceResponse
import android.webkit.WebView
import android.webkit.WebViewClient
import org.json.JSONArray
import org.json.JSONObject
import java.io.ByteArrayInputStream
import java.io.ByteArrayOutputStream
import java.util.concurrent.ConcurrentHashMap

/**
 * Horizon — a tiny native shell around a web UI (assets/index.html).
 * The page is served from a fake https origin so localStorage and canvas colour sampling work,
 * and app icons/banners are streamed to it at https://launcher.local/icon/<package>.
 */
class MainActivity : Activity() {

    companion object { private const val HOST = "launcher.local" }

    private lateinit var web: WebView
    private val iconCache = ConcurrentHashMap<String, ByteArray>()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        web = WebView(this).apply {
            setBackgroundColor(Color.BLACK)
            isFocusable = true
            isFocusableInTouchMode = true
            settings.javaScriptEnabled = true
            settings.domStorageEnabled = true
            settings.useWideViewPort = true
            settings.loadWithOverviewMode = true
            settings.setSupportZoom(false)
            settings.offscreenPreRaster = true
            addJavascriptInterface(Bridge(), "Android")
            webViewClient = object : WebViewClient() {
                override fun shouldInterceptRequest(view: WebView, request: WebResourceRequest): WebResourceResponse? =
                    if (request.url.host == HOST) serve(request.url) else null
            }
        }
        setContentView(web)
        web.loadUrl("https://$HOST/index.html")
        web.requestFocus()
    }

    override fun onResume() {
        super.onResume()
        js("window.onResumeLauncher && onResumeLauncher()")
    }

    // Home pressed while we're already in front → jump back to the first row.
    override fun onNewIntent(intent: Intent?) {
        super.onNewIntent(intent)
        js("window.onHome && onHome()")
    }

    // A launcher never closes on Back.
    @Suppress("OVERRIDE_DEPRECATION", "DEPRECATION")
    override fun onBackPressed() {
        js("window.onBack && onBack()")
    }

    // Remote's menu / options key → pin or unpin the focused app.
    override fun dispatchKeyEvent(event: KeyEvent): Boolean {
        if (event.keyCode == KeyEvent.KEYCODE_MENU) {
            if (event.action == KeyEvent.ACTION_UP) js("window.onMenu && onMenu()")
            return true
        }
        return super.dispatchKeyEvent(event)
    }

    private fun js(code: String) {
        if (::web.isInitialized) web.evaluateJavascript(code, null)
    }

    // ---------- local web server ----------

    private fun serve(uri: Uri): WebResourceResponse? {
        val path = uri.path ?: "/"
        return try {
            if (path.startsWith("/icon/")) {
                val pkg = Uri.decode(path.removePrefix("/icon/"))
                val bytes = iconCache[pkg] ?: renderIcon(pkg)?.also { iconCache[pkg] = it } ?: return null
                WebResourceResponse("image/png", null, ByteArrayInputStream(bytes))
            } else {
                val file = path.trimStart('/').ifEmpty { "index.html" }
                val mime = when {
                    file.endsWith(".html") -> "text/html"
                    file.endsWith(".js") -> "application/javascript"
                    file.endsWith(".css") -> "text/css"
                    file.endsWith(".svg") -> "image/svg+xml"
                    file.endsWith(".png") -> "image/png"
                    else -> "application/octet-stream"
                }
                WebResourceResponse(mime, "utf-8", assets.open(file))
            }
        } catch (e: Exception) {
            null
        }
    }

    /** Prefer the app's TV banner (320x180); fall back to its icon. */
    private fun renderIcon(pkg: String): ByteArray? {
        val pm = packageManager
        val intent = pm.getLeanbackLaunchIntentForPackage(pkg) ?: pm.getLaunchIntentForPackage(pkg)
        val d: Drawable = intent?.let { runCatching { pm.getActivityBanner(it) }.getOrNull() }
            ?: runCatching { pm.getApplicationBanner(pkg) }.getOrNull()
            ?: intent?.let { runCatching { pm.getActivityIcon(it) }.getOrNull() }
            ?: runCatching { pm.getApplicationIcon(pkg) }.getOrNull()
            ?: return null
        val out = ByteArrayOutputStream()
        toBitmap(d).compress(Bitmap.CompressFormat.PNG, 100, out)
        return out.toByteArray()
    }

    private fun toBitmap(d: Drawable): Bitmap {
        if (d is BitmapDrawable && d.bitmap != null) return d.bitmap
        var w = d.intrinsicWidth
        var h = d.intrinsicHeight
        if (w <= 0 || h <= 0) { w = 320; h = 180 }
        val s = maxOf(1f, 320f / maxOf(w, h))
        w = (w * s).toInt(); h = (h * s).toInt()
        val b = Bitmap.createBitmap(w, h, Bitmap.Config.ARGB_8888)
        d.setBounds(0, 0, w, h)
        d.draw(Canvas(b))
        return b
    }

    // ---------- JavaScript bridge (window.Android) ----------

    inner class Bridge {

        @JavascriptInterface
        fun getApps(): String {
            val pm = packageManager
            val out = LinkedHashMap<String, JSONObject>()
            for (cat in listOf(Intent.CATEGORY_LEANBACK_LAUNCHER, Intent.CATEGORY_LAUNCHER)) {
                val q = Intent(Intent.ACTION_MAIN).addCategory(cat)
                for (ri in pm.queryIntentActivities(q, 0)) {
                    val ai = ri.activityInfo
                    val pkg = ai.packageName
                    if (pkg == packageName || out.containsKey(pkg)) continue
                    out[pkg] = JSONObject()
                        .put("pkg", pkg)
                        .put("label", ri.loadLabel(pm).toString())
                        .put("banner", ai.banner != 0 || ai.applicationInfo.banner != 0)
                        .put("tv", cat == Intent.CATEGORY_LEANBACK_LAUNCHER)
                }
            }
            return JSONArray(out.values).toString()
        }

        @JavascriptInterface
        fun launch(pkg: String) = runOnUiThread {
            val pm = packageManager
            val i = pm.getLeanbackLaunchIntentForPackage(pkg) ?: pm.getLaunchIntentForPackage(pkg)
            if (i != null) runCatching { startActivity(i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)) }
        }

        @JavascriptInterface
        fun open(what: String) = runOnUiThread {
            val i = when (what) {
                "settings" -> Intent(Settings.ACTION_SETTINGS)
                "wifi" -> Intent(Settings.ACTION_WIFI_SETTINGS)
                "display" -> Intent(Settings.ACTION_DISPLAY_SETTINGS)
                "apps" -> Intent(Settings.ACTION_APPLICATION_SETTINGS)
                "screensaver" -> Intent(Intent.ACTION_MAIN)
                    .setClassName("com.android.systemui", "com.android.systemui.Somnambulator")
                else -> null
            }
            if (i != null) runCatching { startActivity(i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)) }
        }

        @JavascriptInterface
        fun appInfo(pkg: String) = runOnUiThread {
            runCatching {
                startActivity(
                    Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS, Uri.parse("package:$pkg"))
                        .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                )
            }
        }
    }
}

const fmtDate = new Intl.DateTimeFormat(undefined, {weekday:'long', day:'numeric', month:'long'});
const fmtTime = new Intl.DateTimeFormat(undefined, {hour:'numeric', minute:'2-digit'});
function timeHTML(d){
  let main = '', suf = '';
  for (const p of fmtTime.formatToParts(d)) { if (p.type === 'dayPeriod') suf = p.value; else main += p.value; }
  return esc(main.trim()) + (suf ? `<small>${esc(suf)}</small>` : '');
}
function greeting(h){
  if (h < 5) return 'Late night'; if (h < 12) return 'Good morning';
  if (h < 17) return 'Good afternoon'; if (h < 21) return 'Good evening'; return 'Good night';
}
const ANCH = [
  [0,   ['#03040d','#0a0f2c','#1b1340']],
  [5,   ['#05081a','#1c1a4a','#3b2a5e']],
  [6.3, ['#1d2457','#7a4a86','#f39a6b']],
  [8,   ['#2a5fb8','#4f8cdc','#a9d2ff']],
  [12,  ['#1d58c4','#3f86e0','#8fc4f5']],
  [16,  ['#2a4fa8','#6a7fc8','#f2c39b']],
  [17.6,['#2b2560','#b0577a','#ffa66e']],
  [18.6,['#151a46','#4b2766','#c4515a']],
  [20,  ['#070a22','#121638','#26184a']],
  [24,  ['#03040d','#0a0f2c','#1b1340']]
];
const hex = h => { const n = parseInt(h.slice(1), 16); return [n >> 16, n >> 8 & 255, n & 255]; };
const lerpC = (a, b, t) => { const A = hex(a), B = hex(b); return A.map((v, i) => Math.round(v + (B[i] - v) * t)); };
let night = 1;
function updateSky(h){
  for (let i = 0; i < ANCH.length - 1; i++) {
    const [h0, c0] = ANCH[i], [h1, c1] = ANCH[i + 1];
    if (h >= h0 && h <= h1) {
      const t = (h - h0) / (h1 - h0);
      const cols = c0.map((c, j) => lerpC(c, c1[j], t));
      cols.forEach((c, j) => document.documentElement.style.setProperty('--s' + (j + 1), `rgb(${c.join(',')})`));
      const top = cols[0], lum = 0.2126 * top[0] + 0.7152 * top[1] + 0.0722 * top[2];
      night = Math.max(0, Math.min(1, 1 - lum / 70));
      $('#stars').style.opacity = night;
      return;
    }
  }
}
function updateArc(h){
  const day = h >= SUNRISE && h < SUNSET;
  let t, rem;
  if (day) { t = (h - SUNRISE) / (SUNSET - SUNRISE); rem = SUNSET - h; }
  else {
    const len = 24 - SUNSET + SUNRISE, since = h >= SUNSET ? h - SUNSET : h + 24 - SUNSET;
    t = since / len; rem = len - since;
  }
  const a = Math.PI * t, x = 190 - 170 * Math.cos(a), y = 150 - 120 * Math.sin(a);
  for (const id of ['#orb', '#halo']) { $(id).setAttribute('cx', x.toFixed(1)); $(id).setAttribute('cy', y.toFixed(1)); }
  $('#orb').setAttribute('fill', day ? '#ffd27a' : '#e6ebff');
  $('#prog').style.strokeDasharray = `${t.toFixed(4)} 1`;
  let hh = Math.floor(rem), mm = Math.round((rem - hh) * 60);
  if (mm === 60) { hh++; mm = 0; }
  $('#arcText').textContent = `${day ? 'Sunset' : 'Sunrise'} in ${hh ? hh + 'h ' : ''}${mm}m`;
}
let lastMin = -1;
function tick(){
  const d = new Date(), m = d.getHours() * 60 + d.getMinutes();
  if (m === lastMin) return; lastMin = m;
  const h = m / 60, th = timeHTML(d), ds = fmtDate.format(d);
  $('#time').innerHTML = th; $('#aTime').innerHTML = th;
  $('#date').textContent = ds; $('#aDate').textContent = ds;
  $('#greet').textContent = greeting(d.getHours());
  updateSky(h); updateArc(h);
}

function sample(img){
  try {
    const c = document.createElement('canvas'); c.width = c.height = 24;
    const x = c.getContext('2d'); x.drawImage(img, 0, 0, 24, 24);
    const d = x.getImageData(0, 0, 24, 24).data;
    let r = 0, g = 0, b = 0, w = 0;
    for (let i = 0; i < d.length; i += 4) {
      const R = d[i], G = d[i + 1], B = d[i + 2], A = d[i + 3] / 255;
      const mx = Math.max(R, G, B), mn = Math.min(R, G, B), s = (mx - mn) / 255;
      const wt = A * (0.08 + s * s * 3) * (mx > 40 ? 1 : 0.1);
      r += R * wt; g += G * wt; b += B * wt; w += wt;
    }
    return w ? `rgb(${r / w | 0},${g / w | 0},${b / w | 0})` : null;
  } catch (e) { return null; }
}

function makeCard(a, size){
  const el = document.createElement('div');
  el.className = 'card ' + size; el.dataset.pkg = a.pkg;

  else {
    const src = '/icon/' + encodeURIComponent(a.pkg);
    art = a.banner ? `<div class="art"><img class="ban" src="${src}" alt=""></div>`
                   : `<div class="art tile"><img class="ico" src="${src}" alt=""><span>${esc(a.label)}</span></div>`;
  }
  el.innerHTML = art + `<div class="label">${esc(a.label)}</div><i class="star"></i>`;
  const known = a.mock ? a.mock[0] : accent[a.pkg];
  if (known) el.style.setProperty('--accent', known);
  else {
    const img = el.querySelector('img');
    img.addEventListener('load', () => {
      const c = sample(img); if (!c) return;
      accent[a.pkg] = c; el.style.setProperty('--accent', c);
      if (el.classList.contains('focus')) setGlow(c);
    }, {once: true});
  }
  return el;
}

const tracks = [$('#favs'), $('#all'), $('#dock')];
const rowsEl = [...document.querySelectorAll('.row')];
const rowEls = [[], [], []], off = [0, 0, 0], col = [0, 0, 0];
let fr = 0;

function render(){
  [favs.map(p => byPkg[p]), apps].forEach((list, r) => {
    tracks[r].innerHTML = '';
    rowEls[r] = list.map(a => {
      const el = makeCard(a, r === 0 ? 'lg' : 'md');
      if (r === 1 && favs.includes(a.pkg)) el.classList.add('pinned');
      el.addEventListener('click', () => { focusEl(r, el); activate(); });
      tracks[r].appendChild(el); return el;
    });
  });
  if (!favs.length) tracks[0].innerHTML = '<div class="empty">Hold OK on any app to pin it here</div>';
  tracks[2].innerHTML = '';
  rowEls[2] = DOCK.map(d => {
    const el = document.createElement('div'); el.className = 'pill';
    el.innerHTML = `<svg viewBox="0 0 24 24">${d.icon}</svg><span>${d.label}</span>`;
    el.addEventListener('click', () => { focusEl(2, el); activate(); });
    tracks[2].appendChild(el); return el;
  });
  $('#allCount').textContent = apps.length;
  for (let r = 0; r < 3; r++) col[r] = Math.min(col[r], Math.max(0, rowEls[r].length - 1));

  applyFocus(true);
}
function setGlow(c){ if (c) document.documentElement.style.setProperty('--glow', c); }
function focusEl(r, el){ fr = r; col[r] = rowEls[r].indexOf(el); applyFocus(); }
function applyFocus(instant){
  document.querySelectorAll('.focus').forEach(e => e.classList.remove('focus'));
  rowsEl.forEach((r, i) => r.classList.toggle('active', i === fr));
  const el = rowEls[fr] && rowEls[fr][col[fr]]; if (!el) return;
  el.classList.add('focus');
  const view = el.parentElement.parentElement.clientWidth, m = 96, left = el.offsetLeft, right = left + el.offsetWidth;
  let o = off[fr];
  if (left - o < m) o = left - m;
  if (right - o > view - m) o = right - view + m;
  off[fr] = o = Math.max(0, o);
  const tr = tracks[fr];
  if (instant) tr.style.transition = 'none';
  tr.style.transform = `translateX(${-o}px)`;
  if (instant) { void tr.offsetWidth; tr.style.transition = ''; }
  if (fr < 2) setGlow(el.style.getPropertyValue('--accent'));
}
function bump(dx, dy){
  const el = rowEls[fr][col[fr]];
  if (el && el.animate) el.animate([{translate: '0 0'}, {translate: `${dx * 12}px ${dy * 12}px`}, {translate: '0 0'}], {duration: 260, easing: 'ease-out'});
}
function move(dr, dc){
  if (dc) {
    const c = Math.max(0, Math.min(rowEls[fr].length - 1, col[fr] + dc));
    if (c === col[fr]) return bump(dc, 0);
    col[fr] = c;
  }
  if (dr) {
    let r = fr + dr;
    while (r >= 0 && r < 3 && !rowEls[r].length) r += dr;
    if (r < 0 || r > 2) return bump(0, dr);
    const cur = rowEls[fr][col[fr]].getBoundingClientRect(), x = cur.left + cur.width / 2;
    let best = 0, bd = Infinity;
    rowEls[r].forEach((e, i) => { const b = e.getBoundingClientRect(), d = Math.abs(b.left + b.width / 2 - x); if (d < bd) { bd = d; best = i; } });
    col[r] = best; fr = r;
  }
  applyFocus();
}

function togglePin(){

  if (i >= 0) { favs.splice(i, 1); toast(`Removed ${a.label} from favourites`); }
  else { favs.push(pkg); toast(`★ Pinned ${a.label}`); }
  saveFavs(favs); render();
}

/* ====== remote / keyboard ====== */
const OK = k => k === 'Enter' || k === ' ' || k === 'NumpadEnter';
let downAt = 0, longDone = false;
addEventListener('keydown', e => {
  if (ambient) { e.preventDefault(); swallow = true; exitAmbient(); return; }
  poke();
  const k = e.key;
  if (k === 'ArrowLeft') move(0, -1);
  else if (k === 'ArrowRight') move(0, 1);
  else if (k === 'ArrowUp') move(-1, 0);
  else if (k === 'ArrowDown') move(1, 0);
  else if (OK(k)) {
    if (!e.repeat) { downAt = Date.now(); longDone = false; }
    else if (!longDone && downAt && Date.now() - downAt > 600) { longDone = true; togglePin(); }
  }
  else if (k === 'ContextMenu' || k === 'm' || k === 'M') togglePin();
  else return;
  e.preventDefault();
});
addEventListener('keyup', e => {
  if (swallow) { swallow = false; downAt = 0; return; }
  if (OK(e.key)) { if (downAt && !longDone) activate(); downAt = 0; }
});
document.addEventListener('mousemove', () => { if (ambient) exitAmbient(); else poke(); });
